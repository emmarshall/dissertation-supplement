Exploratory Analyses

October 10, 2026
Emma Marshall, J.D., M.A.
University of Nebraska-Lincoln

This chapter reports the exploratory analyses that support the primary Pilot 2 tests. They include the analyses based on causal model theory, the attitude analyses, and the analyses of later judgments.

r
p2_a45_tbl <- p2_a4 |>
  mutate(
    contrast = dplyr::recode(contrast,
      "RH3.1 indirect a*b (registered fit)" = "RH3.1 observed-composite companion indirect",
      .default = contrast),
    est = paste0(zr_fmt(estimate, 3),
                 ifelse(!is.na(hdi_low) & !is.na(hdi_high) &
                        ((hdi_low > 0 & hdi_high > 0) | (hdi_low < 0 & hdi_high < 0)),
                        "*", "")),
    ci  = mapply(fmt_ci, hdi_low, hdi_high, digits = 3),
    pp  = vapply(p_gt_0, fmt_pp, character(1))
  ) |>
  select(contrast, est, ci, pp)

p2_a45_tbl <- p2_a45_tbl |> dplyr::rename(
  `Indirect route` = contrast,
  `Indirect` = est,
  `95% HDI` = ci,
  `P(indirect > 0)` = pp
)
dissAPA::apa_render(p2_a45_tbl, engine = "auto", fontsize = 0.9)

p2_rq5_tbl <- bind_rows(
  p2_c6 |> mutate(family = "Attitude moderation",  label = paste(dv, term, sep = " · ")) |>
           select(family, label, estimate, hdi_low, hdi_high, p_gt_0),
  p2_c8 |> mutate(family = "Hostile/benevolent-sexism parallel mediation", label = contrast) |>
           select(family, label, estimate, hdi_low, hdi_high, p_gt_0)
) |>
  mutate(
    est = paste0(zr_fmt(estimate, 3),
                 ifelse(!is.na(hdi_low) & !is.na(hdi_high) &

                        "*", "")),
    ci  = mapply(fmt_ci, hdi_low, hdi_high, digits = 3),
    pp  = vapply(p_gt_0, fmt_pp, character(1))
  ) |>
  select(family, label, est, ci, pp)

p2_rq5_tbl <- p2_rq5_tbl |> dplyr::rename(
