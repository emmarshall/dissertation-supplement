Validation and Reliability of the Narrative Coding

The narrative codes were assigned in three phases. In the first two phases, automated coding accepted a code when its certainty met a calibrated cutoff. Codes below the cutoff went to human coders in Phase 3. Table P1.2 shows the cutoff for each phase, and Table P1.3 shows the inter-rater reliability of the holistic Phase 3 codes. The keyness analyses in Table P1.4 and Table P1.5 serve as a convergent check on the coding. They show the words that most set each crime category apart from the others.

Crime-Category Pairwise Comparisons

I followed up the H3b within-subjects ANOVA with pairwise comparisons of mean severity across the four crime categories. The comparisons used Fisher’s least significant difference procedure (Levin et al., 1994). With four groups that procedure does not hold the familywise error rate at .05, so Table P1.6 also shows Tukey-adjusted p values.

r
if (is.null(p1_rh32_pairs)) {
  stop("rq3_rh32_pairwise.csv is missing: render analysis/pilot01/06_rq3_severity.qmd first.",
       call. = FALSE)
}
rh32_omni <- dplyr::distinct(p1_rh32_pairs, omnibus_F, omnibus_df1, omnibus_df2, omnibus_p)
stopifnot(
  "rq3_rh32_pairwise.csv: expected six pairs sharing one omnibus test" =
    nrow(p1_rh32_pairs) == 6L && nrow(rh32_omni) == 1L
)

rh32_fmt_df <- function(x) {
  ifelse(abs(x - round(x)) < 1e-8, sprintf("%.0f", x), sprintf("%.2f", x))
}
rh32_fmt_p <- function(p) ifelse(p < .001, "< .001", sub("^0", "", sprintf("%.3f", p)))

rh32_pairs_tbl <- p1_rh32_pairs |>
  transmute(
    `Comparison`        = sub(" - ", " − ", contrast, fixed = TRUE),
    `Difference`        = zr_fmt(estimate, 2),
    `t`                 = zr_plain(t, 2),
    `df`                = rh32_fmt_df(df),
    `p`                 = rh32_fmt_p(p_own),
    `Significant (LSD)` = ifelse(lsd_significant, "Yes", "No"),
    `p (Tukey)`         = rh32_fmt_p(p_tukey)
  )

# The omnibus test the LSD procedure is gated on, as the notebook saved it. A
# fractional numerator df is the Greenhouse-Geisser correction.
rh32_omni_txt <- paste0(
  "F(", rh32_fmt_df(rh32_omni$omnibus_df1), ", ", rh32_fmt_df(rh32_omni$omnibus_df2),
  ") = ", sprintf("%.2f", rh32_omni$omnibus_F), ", p ",
  ifelse(rh32_omni$omnibus_p < .001, "< .001",
         paste0("= ", sub("^0", "", sprintf("%.3f", rh32_omni$omnibus_p)))),
  if (abs(rh32_omni$omnibus_df1 - round(rh32_omni$omnibus_df1)) > 1e-8)
    ", Greenhouse–Geisser corrected" else ""
)
rh32_pairs_note <- paste0(
  if (rh32_omni$omnibus_p < .05) {
    paste0("Following a significant omnibus test of crime category (", rh32_omni_txt,
           "), the comparisons used Fisher's least significant difference (LSD) ",
           "procedure: a pair is significant when its own, unadjusted p is below .05. ")
  } else {
    paste0("The omnibus test of crime category was not significant (", rh32_omni_txt,
           "), so by Fisher's least significant difference (LSD) procedure no pair ",
           "is significant; each pair's own, unadjusted p is shown. ")
  },
  # tinytable prints a Note as written, so a Pandoc citation key would show
  # literally here; the section's lead sentence carries the linked citation.
  "Because with four categories that procedure does not hold the familywise ",
  "error rate at .05, Tukey-adjusted p values are shown beside it (Levin et al., 1994)."
)

dissAPA::apa_render(rh32_pairs_tbl, engine = "auto", note = rh32_pairs_note, fontsize = 0.9)

Exploratory and Legal-Accuracy Analyses

The tables below report the exploratory regressions and the accuracy of participants’ legal judgments for each scenario. I describe the legal-accuracy results in the next chapter of this part.

r
exp1_term <- function(x) {
  dplyr::recode(
    x,
    "(Intercept)"                              = "Intercept",
    "aas_z"                                    = "Abortion attitudes (z)",
    "is_crime_in_state"                        = "State law (crime in state)",
    "judgment_typenorm"                        = "Judgment type: Normative",
    "aas_z:is_crime_in_state"                  = "Attitudes x State law",
    "aas_z:judgment_typenorm"                  = "Attitudes x Judgment type",
    "is_crime_in_state:judgment_typenorm"      = "State law x Judgment type",
    "aas_z:is_crime_in_state:judgment_typenorm" = "Attitudes x State law x Judgment type",
    .default = x
  )
}

exp1_input <- p1_exp1 |>
  filter(Effects == "fixed") |>
  mutate(
    Parameter = factor(
      Parameter,
      levels = c("(Intercept)", "aas_z", "is_crime_in_state",
                 "judgment_typenorm", "aas_z:is_crime_in_state",
                 "aas_z:judgment_typenorm",
                 "is_crime_in_state:judgment_typenorm",
                 "aas_z:is_crime_in_state:judgment_typenorm")
    )
  ) |>
  arrange(Parameter) |>
  mutate(Parameter = exp1_term(as.character(Parameter)))

exp1_note <- paste0(
  "OR = odds ratio (exponentiated fixed-effect coefficient) from a mixed-effects ",
  "logistic regression with crossed participant and scenario random intercepts; ",
  "95% CI is the Wald interval on the OR. Reference levels: conduct not currently a ",
  "crime in the participant's state, and a descriptive judgment."
)

# OR contract (Term | OR | 95% CI | z | p) via the dissAPA house function; the
# parameters() export is already exponentiated, so no further transform is needed.
dissAPA::apa_render(dissAPA::apa_glmm_or(exp1_input), engine = "auto", note = exp1_note)
r
exp2_term <- function(x) {
  dplyr::recode(
    x,
    "asi_benevolent_mean" = "Benevolent sexism",
    "asi_hostile_mean"    = "Hostile sexism",
    "religiosity_mean"    = "Religiosity",
    .default = x
  )
}

exp2_input <- p1_exp2 |>
  filter(Parameter != "(Intercept)") |>
  mutate(
    outcome   = factor(outcome,
                       levels = c("severity", "guilt", "punishment"),
                       labels = c("Severity", "Guilt", "Punishment")),
    Parameter = factor(Parameter,
                       levels = c("asi_benevolent_mean", "asi_hostile_mean",
                                  "religiosity_mean"))
  ) |>
  arrange(outcome, Parameter) |>
  mutate(Parameter = exp2_term(as.character(Parameter)))

exp2_note <- paste0(
  "β = standardized regression coefficient from a multiple regression of the ",
  "participant-level mean rating on benevolent sexism, hostile sexism, and ",
  "religiosity (entered simultaneously); 95% CI = Wald interval; t-tests use 77 ",
  "residual df."
)

# Standardized-beta regression contract (Outcome | Predictor | β | 95% CI | t | p) via
# the dissAPA house function; values come straight from p1_exp2 (parameters() output).
dissAPA::apa_render(
  dissAPA::apa_regression(exp2_input, group = "outcome", term = "Predictor",
                          estimate = "β"),
  engine = "auto", note = exp2_note
)
